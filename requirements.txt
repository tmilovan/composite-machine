# composite-machine — pinned to the versions this project is developed and
# tested against.  Python 3.11.4.
#
#     python3 -m venv composite-env
#     source composite-env/bin/activate
#     pip install -r requirements.txt
#
# The shared virtualenv this repo is usually run from carries ~43 packages,
# most of them belonging to neighbouring experiments (elkai, flopy, ortools
# and their dependencies).  `pip freeze` would capture all of those, so this
# file lists only what the project's own code imports, established by parsing
# every .py in composite/, tests/ and demos/ rather than by reading the env.

# ---- runtime -----------------------------------------------------------
# The only hard dependency.  Every backend uses it for the coefficient and
# dimension arrays, and resummation/singularity use it for roots and the
# least-squares fit behind the differential approximants.
numpy==1.26.4

# ---- tests and demos ---------------------------------------------------
# pytest runs the suites through tests/test_suites.py.  Most suites are
# standalone scripts that exit on their own tally, so pytest is a runner
# rather than a framework, and they can be invoked directly without it.
pytest==9.1.1

# mpmath is the INDEPENDENT ORACLE, not a helper: the derivative, transseries
# and forensics suites pin their references from it at 50-60 decimal digits
# and re-check against it when it is importable.  Nothing in composite/ imports
# it, so the library installs and runs without it -- only the checking does.
mpmath==1.3.0

# ---- packaging ---------------------------------------------------------
# setup.py only.  Not needed to run or test.
setuptools==82.0.0
